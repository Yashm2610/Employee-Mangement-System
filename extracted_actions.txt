    const empComponentsData = {

            "{{ emp_id }}": [
                {% for c in comps %}
                    {"name": "{{ c.component_name|replace('"', '\\"') }}", "code": {{ c.component_code }}, "amount": {{ c.amount }} }{% if not loop.last %},{% endif %}
                {% endfor %}
            ]{% if not loop.last %},{% endif %}

    };

    function openEditModal(empId, empName, bankName, bankAcc, ifsc, basicSalary) {
        document.getElementById('modalEmpId').value = empId;
        document.getElementById('modalEmpName').textContent = empName;
        document.getElementById('modalBankName').value = bankName;
        document.getElementById('modalBankAcc').value = bankAcc;
        document.getElementById('modalIfsc').value = ifsc;
        document.getElementById('modalBasicSalary').value = basicSalary;
        
        const container = document.getElementById('componentsContainer');
        container.innerHTML = ''; // clear existing
        
        const comps = empComponentsData[empId] || [];
        if (comps.length > 0) {
            comps.forEach(c => addComponentRow(c.name, c.code, c.amount));
        } else {
            // Provide at least one empty row
            addComponentRow();
        }
        
        new bootstrap.Modal(document.getElementById('editFinancialsModal')).show();
    }
    
    function addComponentRow(name = '', code = 1, amount = '') {
        const container = document.getElementById('componentsContainer');
        const row = document.createElement('div');
        row.className = 'input-group input-group-sm component-row';
        row.innerHTML = `
            <input type="text" class="form-control w-40" name="component_name[]" placeholder="Component Name (e.g. HRA)" value="${name}" required>
            <select class="form-select" name="component_code[]">
                <option value="1" ${code == 1 ? 'selected' : ''}>Allowance (+)</option>
                <option value="2" ${code == 2 ? 'selected' : ''}>Deduction (-)</option>
            </select>
            <input type="number" step="0.01" class="form-control w-25" name="component_amount[]" placeholder="Amount (Γé╣)" value="${amount}" required>
            <button class="btn btn-outline-danger" type="button" onclick="this.parentElement.remove()">
                <i class="bi bi-trash"></i>
            </button>
        `;
        container.appendChild(row);
    }

    function filterFinancials() {
        const input = document.getElementById("financialSearch").value.toLowerCase();
        const table = document.querySelector(".custom-table");
        const tr = table.getElementsByTagName("tr");
        
        for (let i = 1; i < tr.length; i++) {
            if (tr[i].cells.length < 3) continue; // Skip 'No records' row
            
            const empId = tr[i].cells[0].textContent.toLowerCase();
            const empName = tr[i].cells[1].textContent.toLowerCase();
            const bankDetails = tr[i].cells[2].textContent.toLowerCase();
            
            if (empId.indexOf(input) > -1 || empName.indexOf(input) > -1 || bankDetails.indexOf(input) > -1) {
                tr[i].style.display = "";
            } else {
                tr[i].style.display = "none";
            }
        }
    }
